<script>
document.addEventListener('alpine:init', () => {
    Alpine.data('studioPortail', (init) => ({
        tab: init.tab,
        mode: 'mobile',
        envoi: false,
        partant: false,
        form: init.form,
        initial: '',
        logoUrl: init.logoUrl,
        logoInitial: init.logoUrl,
        logoNom: '',
        logoErreur: '',
        supprimerLogo: false,
        tarif: init.tarif,
        port: init.port,
        echelle: 1,

        get cadre() {
            return this.mode === 'mobile' ? { w: 399, h: 764 } : { w: 1000, h: 680 };
        },
        get rayonPreset() {
            const r = Number(this.form.bouton_rayon);
            return r <= 8 ? 'carre' : (r >= 40 ? 'pilule' : 'doux');
        },
        get urlAffichee() {
            return 'http://' + (this.form.portail_domaine || 'portail') + (this.port !== 80 ? ':' + this.port : '') + '/portail';
        },
        get fondEcran() {
            const f = this.form;
            return {
                fontFamily: f.police,
                background: f.fond_type === 'gradient'
                    ? 'radial-gradient(120% 90% at 50% 0, ' + f.fond_couleur_2 + ' 0, transparent 70%), ' + f.fond_couleur_1
                    : f.fond_couleur_1,
            };
        },
        get dirty() {
            return JSON.stringify(this.form) !== this.initial || this.logoNom !== '' || this.supprimerLogo;
        },

        init() {
            this.initial = JSON.stringify(this.form);
            this.$nextTick(() => {
                this.ajuster();
                if (window.ResizeObserver && this.$refs.scene) {
                    new ResizeObserver(() => this.ajuster()).observe(this.$refs.scene);
                }
            });
            this.$watch('mode', () => this.$nextTick(() => this.ajuster()));
            window.addEventListener('beforeunload', (e) => {
                if (this.dirty && !this.partant) { e.preventDefault(); e.returnValue = ''; }
            });
        },

        ajuster() {
            const s = this.$refs.scene;
            if (!s || !s.clientWidth) return;
            this.echelle = Math.min(1, s.clientWidth / this.cadre.w);
        },

        changerLogo(e) {
            const f = e.target.files[0];
            if (!f) return;
            if (f.size > 2 * 1024 * 1024) {
                this.logoErreur = 'Image trop lourde (2 Mo au plus).';
                e.target.value = '';
                return;
            }
            this.logoErreur = '';
            this.logoNom = f.name;
            this.logoUrl = URL.createObjectURL(f);
            this.supprimerLogo = false;
        },

        retirerLogo() {
            this.logoUrl = '';
            this.logoNom = '';
            this.logoErreur = '';
            this.supprimerLogo = true;
            if (this.$refs.logoInput) this.$refs.logoInput.value = '';
        },

        annuler() {
            this.form = JSON.parse(this.initial);
            this.logoUrl = this.logoInitial;
            this.logoNom = '';
            this.logoErreur = '';
            this.supprimerLogo = false;
            if (this.$refs.logoInput) this.$refs.logoInput.value = '';
        },
    }));
});
</script>

                const t = document.createElement('textarea');
                t.value = texte; t.style.position = 'fixed'; t.style.opacity = '0';

    private function traiter(Request $request, bool $diffuser, bool $redemarrer)
    {
        $d = $request->validate($this->regles(), self::MESSAGES, self::ATTRIBUTS);

        $valeurs = [];
        foreach (PortailConfig::CHAMPS as $cle) {
            $valeurs[$cle] = $d[$cle] ?? ($cle === 'annonce_active' ? '0' : '');
        }

        $ancienLogo  = PortailParametre::get('logo');
        $nouveauLogo = $ancienLogo;
        if ($request->hasFile('logo')) {
            $nouveauLogo = $request->file('logo')->store('portail', 'public');
        } elseif ($request->boolean('supprimer_logo')) {
            $nouveauLogo = null;
        }
        $valeurs['logo'] = $nouveauLogo;

        PortailConfig::enregistrer($valeurs);

        if ($ancienLogo && $ancienLogo !== $nouveauLogo) {
            Storage::disk('public')->delete($ancienLogo);
        }

        if (filled($d['portail_domaine'] ?? null)) {
            CyberSetting::setReseau('portail_domaine', strtolower(trim($d['portail_domaine'])));
        }
        CyberSetting::setReseau('portail_ip_serveur', filled($d['portail_ip_serveur'] ?? null) ? $d['portail_ip_serveur'] : null);

        $routeurModifie = false;
        if ($request->boolean('efface_routeur')) {
            foreach (['mikrotik_host', 'mikrotik_port', 'mikrotik_user', 'mikrotik_password'] as $k) {
                CyberSetting::setReseau($k, null);
            }
            $routeurModifie = true;
        } else {
            foreach (['mikrotik_host', 'mikrotik_port', 'mikrotik_user', 'mikrotik_password'] as $k) {
                if (filled($d[$k] ?? null)) {
                    CyberSetting::setReseau($k, (string) $d[$k]);
                    $routeurModifie = true;
                }
            }
        }
        if ($routeurModifie) {
            Cache::forget('mikrotik.statut');
            Cache::forget('wifi_sync_etat_routeur');
            Cache::forget('wifi_sync_derniere_tentative');
        }

        $retour = redirect()->route('portail.admin')->with('onglet', $d['onglet'] ?? 'general');

        if (!$diffuser) {
            return $retour->with('success', 'Réglages enregistrés. Vos clients voient déjà les nouveaux textes et couleurs.');
        }

        $r = $this->synchroniserRouteur($request, $redemarrer);

        return $r['ok']
            ? $retour->with('success', 'Réglages enregistrés. ' . $r['message'])
            : $retour->with('success', 'Réglages enregistrés : vos clients voient déjà les nouveaux textes et couleurs.')
                     ->with('error', $r['message']);
    }

    private function synchroniserRouteur(Request $request, bool $redemarrer): array
    {
        $mt = new MikroTikService();

        if (!$mt->estConfigure()) {
            return $this->noter(false, null, 'Aucun routeur n\'est configuré (onglet Réseau, « Accès au routeur »).');
        }

        $ip = $this->ipServeur($mt);
        if ($ip === null) {
            return $this->noter(false, null, 'L\'adresse de ce PC sur le réseau n\'a pas été trouvée. Saisissez-la dans l\'onglet Réseau (« Forcer une adresse »).');
        }

        $port    = (string) ($request->getPort() ?: 80);
        $domaine = CyberSetting::reseau('portail_domaine', PortailConfig::DOMAINE_DEFAUT);
        $url     = 'http://' . $ip . ($port !== '80' ? ':' . $port : '') . '/portail';

        try {
            $mt->nettoyerReglesObsolètesPortail();
            $mt->configurerWalledGarden($ip, $port);
            $mt->verifierEtAjusterDhcpDns();

            if ($domaine) {
                $mt->configurerDnsLocal($domaine, $ip);
            }

            if (!$mt->installerRedirectionHotspot($url)) {
                throw new RuntimeException('Le fichier de redirection n\'a pas pu être écrit sur le routeur.');
            }

            if ($redemarrer) {
                $mt->redemarrerHotspotEtPurger();
            } else {
                $mt->viderCacheDns();
            }
        } catch (\Throwable $e) {
            Log::warning('Portail : synchronisation du routeur échouée', ['error' => $e->getMessage()]);

            return $this->noter(false, $ip, $this->messageRouteur($e));
        }

        $message = "Routeur synchronisé : les clients sont redirigés vers {$url}"
            . ($domaine ? " (nom « {$domaine} » enregistré dans le routeur)." : '.')
            . ($redemarrer ? ' Le HotSpot a été redémarré.' : '');

        return $this->noter(true, $ip, $message);
    }

    private function noter(bool $ok, ?string $ip, string $message): array
    {
        CyberSetting::setReseau('portail_sync_at', now()->toIso8601String());
        CyberSetting::setReseau('portail_sync_ok', $ok ? '1' : '0');
        CyberSetting::setReseau('portail_sync_message', $ok ? null : mb_substr($message, 0, 250));
        if ($ok && $ip) {
            CyberSetting::setReseau('portail_ip_synchronisee', $ip);
        }

        Cache::forget('mikrotik.statut');
        Cache::forget('wifi_sync_etat_routeur');
        Cache::forget('wifi_sync_derniere_tentative');

        return ['ok' => $ok, 'message' => $message];
    }

    private function messageRouteur(\Throwable $e): string
    {
        $brut = $e->getMessage();

        return match (true) {
            (bool) preg_match('/impossible|timeout|ferm|refused/i', $brut)

                => 'Le routeur refuse l\'utilisateur ou le mot de passe (onglet Réseau, « Accès au routeur »).',
            default => 'Le routeur a refusé une commande : ' . mb_substr($brut, 0, 160),
        };
    }

    private function etatSynchro(?string $ipActuelle): array
    {
        $date = null;
        try {
            $at   = CyberSetting::reseau('portail_sync_at');
            $date = $at ? Carbon::parse($at) : null;
        } catch (\Throwable) {
        }

        $ipSync = CyberSetting::reseau('portail_ip_synchronisee');

        return [
            'jamais'          => $date === null,
            'ok'              => CyberSetting::reseau('portail_sync_ok') === '1',
            'il_y_a'          => $date?->locale('fr')->diffForHumans(),
            'message'         => CyberSetting::reseau('portail_sync_message'),
            'ip_synchronisee' => $ipSync,
            'ip_changee'      => $ipSync && $ipActuelle && $ipSync !== $ipActuelle,
        ];
    }

    private function ipServeur(MikroTikService $mt): ?string
    {
        $manuelle = CyberSetting::reseau('portail_ip_serveur');
        if ($manuelle && $this->ipUtilisable($manuelle)) {
            return $manuelle;
        }

        // 1. Détection locale de l'IP faisant face au routeur MikroTik
        try {
            $ip = $mt->detecterIpLocaleVersRouteur();
            if ($ip && $this->ipUtilisable($ip)) {
                return $ip;
            }
        } catch (\Throwable) {
        }

        // 2. Repli : route principale du système
        $ip = ReseauService::ipPrincipale();

        return $ip && $this->ipUtilisable($ip) ? $ip : null;
    }

    private function ipUtilisable(?string $ip): bool
    {
        return $ip
            && filter_var($ip, FILTER_VALIDATE_IP, FILTER_FLAG_IPV4)
            && !str_starts_with($ip, '127.')
            && !str_starts_with($ip, '169.254.')
            && $ip !== '0.0.0.0';
    }

    private function macValide(?string $mac): ?string
    {
        if (!$mac) return null;
        $mac = strtoupper(str_replace('-', ':', trim($mac)));
        return preg_match('/^([0-9A-F]{2}:){5}[0-9A-F]{2}$/', $mac) ? $mac : null;
    }

    private function traduireErreur(?string $brut): ?string
    {
        if (!$brut) return null;
        return match (true) {
            (bool) preg_match('/invalid (username|password)/i', $brut) => 'Identifiant ou mot de passe incorrect.',
            (bool) preg_match('/(uptime|time) limit/i', $brut)         => 'Le temps accordé à ce ticket est épuisé.',
            (bool) preg_match('/(bytes|data|traffic) limit/i', $brut)  => 'Le quota de données de ce ticket est épuisé.',
            (bool) preg_match('/already|simultaneous|active/i', $brut) => 'Ce compte est déjà utilisé sur un autre appareil.',
            (bool) preg_match('/disabled/i', $brut)                    => 'Ce compte est désactivé. Veuillez voir la caisse.',
            default => 'Connexion refusée. Vérifiez vos identifiants.',
        };
    }

    private function lienLoginAutorise(string $url): bool
    {
        $p = parse_url($url);
        if (!$p || !in_array($p['scheme'] ?? '', ['http', 'https'], true) || empty($p['host'])) {
            return false;
        }

        $hote = $p['host'];

        if (filter_var($hote, FILTER_VALIDATE_IP)) {
            return !filter_var($hote, FILTER_VALIDATE_IP, FILTER_FLAG_NO_PRIV_RANGE | FILTER_FLAG_NO_RES_RANGE);
        }

        return !str_contains($hote, '.') || (bool) preg_match('/\.(local|lan)$/i', $hote);
    }

    public function apiStats(Request $request): JsonResponse
    {
        $macNorm = $this->macValide($request->query('mac'));
        $ip = $request->ip();

        $session = Session::query()
            ->where('type_session', 'wifi')
            ->where(function ($q) use ($macNorm, $ip) {
                if ($macNorm) {
                    $q->where('client_mac', $macNorm);
                } else {
                    $q->where('client_ip', $ip);
                }
            })
            ->orderByDesc('id')
            ->first();

        $tarif = $session?->tarif ?: $this->tarifService->tarifWifi();
        $parMin = max(1, (int) ($tarif->montant_par_minute ?? 20));
        $minimum = (int) ($tarif->montant_minimum ?? 300);

        if (!$session) {
            try {
                $filtre = $macNorm ? ['mac-address' => $macNorm] : ['address' => $ip];
                $active = Cache::remember("mt_active_{$macNorm}_{$ip}", 2, function () use ($filtre) {
                    return $this->mikroTik->queryFiltered('/ip/hotspot/active/print', $filtre)[0] ?? null;
                });

                if (!$active) {
                    return response()->json(['connected' => false, 'terminee' => false])->header('Cache-Control', 'no-store');
                }

                $sec = $this->parseUptime($active['uptime'] ?? '0s');
                $min = intdiv($sec, 60);
                $montant = $tarif ? $this->tarifService->montantPourMinutes($min, $tarif) : max($min * $parMin, $minimum);

                return response()->json([
                    'connected'        => true,
                    'en_pause'         => false,
                    'terminee'         => false,
                    'uptime'           => $sec,
                    'bytes_down'       => (int) ($active['bytes-out'] ?? 0),
                    'bytes_up'         => (int) ($active['bytes-in'] ?? 0),
                    'montant'          => $montant,
                    'minimum_applique' => ($min * $parMin) < $minimum,
                    'tarif_minute'     => $parMin,
                    'tarif_minimum'    => $minimum,
                ])->header('Cache-Control', 'no-store');
            } catch (\Throwable $e) {
                return response()->json(['error' => 'Routeur injoignable'], 503)->header('Cache-Control', 'no-store');
            }
        }

        $estTermine = $session->estTermine();
        $estConnecte = $session->estConnecte();
        $estEnPause = $session->estEnPause() || $session->estDeconnecte();

        $secondes = $session->secondesConsommees();
        $minutes = intdiv($secondes, 60);

        $montant = $tarif
            ? $this->tarifService->montantPourMinutes($minutes, $tarif)
            : max($minutes * $parMin, $minimum);

        return response()->json([
            'connected'        => $estConnecte,
            'en_pause'         => $estEnPause,
            'terminee'         => $estTermine,
            'uptime'           => $secondes,
            'bytes_down'       => (int) $session->volume_entree,
            'bytes_up'         => (int) $session->volume_sortie,
            'montant'          => $montant,
            'minimum_applique' => $minimum > 0 && ($minutes * $parMin) < $minimum,
            'tarif_minute'     => $parMin,
            'tarif_minimum'    => $minimum,
        ])->header('Cache-Control', 'no-store');
    }

    private function parseUptime(string $uptime): int
    {
        $t = 0;
        if (preg_match('/(\d+)w/', $uptime, $m)) $t += (int) $m[1] * 604800;
        if (preg_match('/(\d+)d/', $uptime, $m)) $t += (int) $m[1] * 86400;
        if (preg_match('/(\d+)h/', $uptime, $m)) $t += (int) $m[1] * 3600;
        if (preg_match('/(\d+)m(?!s)/', $uptime, $m)) $t += (int) $m[1] * 60;
        if (preg_match('/(\d+)s/', $uptime, $m)) $t += (int) $m[1];
        return $t;
    }

<?php

namespace App\Services;

/** Adresse du serveur sur le réseau local (jamais « localhost »). */
class ReseauService
{
    /** Adresse de la carte réseau qui sert à joindre le réseau (route par défaut). */
    public static function ipPrincipale(): ?string
    {
        try {
            $s = @stream_socket_client('udp://192.0.2.1:53', $errno, $errstr, 1);
            if ($s) {
                $nom = (string) stream_socket_get_name($s, false);
                fclose($s);
                $ip = explode(':', $nom)[0] ?? '';
                if (self::valide($ip)) {
                    return $ip;
                }
            }
        } catch (\Throwable) {
        }

        return self::toutes()[0] ?? null;
    }

    /** Toutes les adresses IPv4 de l'ordinateur (utile s'il a plusieurs cartes réseau). */
    public static function toutes(): array
    {
        $liste = [];

        try {
            if (function_exists('net_get_interfaces')) {
                foreach ((array) @net_get_interfaces() as $nom => $carte) {
                    if (isset($carte['up']) && $carte['up'] === false) {
                        continue;
                    }
                    if (preg_match('/vEthernet|VirtualBox|VMware|Hyper-V|WSL|Loopback|Bluetooth|Tailscale|ZeroTier/i', (string) $nom)) {
                        continue;   // cartes virtuelles : jamais l'adresse que les postes doivent utiliser
                    }
                    foreach ((array) ($carte['unicast'] ?? []) as $adresse) {
                        $ip = (string) ($adresse['address'] ?? '');
                        if (self::valide($ip)) {
                            $liste[] = $ip;
                        }
                    }
                }
            }
        } catch (\Throwable) {
        }

        try {
            foreach ((array) @gethostbynamel((string) gethostname()) as $ip) {
                if (self::valide($ip)) {
                    $liste[] = $ip;
                }
            }
        } catch (\Throwable) {
        }

        return array_values(array_unique($liste));
    }

    private static function valide(string $ip): bool
    {
        return filter_var($ip, FILTER_VALIDATE_IP, FILTER_FLAG_IPV4)
            && !str_starts_with($ip, '127.')
            && !str_starts_with($ip, '169.254.');
    }

    /**
     * URL que les postes doivent utiliser pour joindre ce serveur.
     *
     *  - Port : celui par lequel la page a été ouverte. C'est, par définition, un port où le serveur répond réellement
     *    (80 avec Apache, 8000 avec « artisan serve », ou n'importe quel autre).
     *  - Adresse : si la page a été ouverte avec une adresse IP de CET ordinateur (http://192.168.1.10:8000),
     *    on garde celle-là (cas de plusieurs cartes réseau) ; sinon (localhost, nom de l'ordinateur) on prend
     *    l'adresse de la carte réseau par défaut.
     */
    public static function urlServeur(?int $port = null): string
    {
        $requete = function_exists('request') ? request() : null;

        $port = ($port !== null && $port > 0) ? $port : (int) ($requete?->getPort() ?: 80);
        $hote = (string) ($requete?->getHost() ?? '');

        $ip = null;
        if ($hote !== '' && filter_var($hote, FILTER_VALIDATE_IP, FILTER_FLAG_IPV4) && self::valide($hote)
            && in_array($hote, self::toutes(), true)) {
            $ip = $hote;
        }
        $ip ??= self::ipPrincipale() ?? '127.0.0.1';

        return 'http://' . $ip . ($port === 80 ? '' : ':' . $port);
    }
}

    private function send(array $words): array
    {
        $this->login();
        $reply = $this->connection()->sendSentence($words);

        if ($this->replyHasTrap($reply)) {
            $msg = $this->extractTrapMessage($reply);
            $this->disconnect();
            throw new RuntimeException('MikroTik ' . ($words[0] ?? '?') . ' : ' . $msg);
        }

        $records = [];
        foreach ($reply as $item) {
            if ($item['type'] === '!re') {
                $records[] = $item['attributes'];
            }
        }
        $this->disconnect();
        return $records;
    }

    public static function formatUptime(int $minutes): string

    public function detecterIpLocaleVersRouteur(): ?string
    {
        $hoteRouteur = $this->config['host'] ?? null;
        if (!$hoteRouteur) return null;

        $port = (int) ($this->config['port'] ?? 8728);

        // 1. Tente par flux natif UDP (standard PHP sans extension optionnelle)
        try {
            $sock = @stream_socket_client("udp://{$hoteRouteur}:{$port}", $errno, $errstr, 1);
            if ($sock) {
                $nom = (string) @stream_socket_get_name($sock, false);
                @fclose($sock);
                $ip = explode(':', $nom)[0] ?? '';
                if (!empty($ip) && $ip !== '0.0.0.0' && !str_starts_with($ip, '127.') && !str_starts_with($ip, '169.254.')) {
                    return $ip;
                }
            }
        } catch (\Throwable) {}

        // 2. Tente par socket_create si l'extension php_sockets est activée
        if (function_exists('socket_create')) {
            $sock = @socket_create(AF_INET, SOCK_DGRAM, SOL_UDP);
            if ($sock) {
                if (@socket_connect($sock, $hoteRouteur, $port)) {
                    @socket_getsockname($sock, $ipLocale);
                    @socket_close($sock);
                    if (!empty($ipLocale) && $ipLocale !== '0.0.0.0' && !str_starts_with($ipLocale, '127.') && !str_starts_with($ipLocale, '169.254.')) {
                        return $ipLocale;
                    }
                }
                @socket_close($sock);
            }
        }

        // 3. Repli : trouve l'adresse IPv4 locale du même sous-réseau
        $partiesRouteur = explode('.', $hoteRouteur);
        if (count($partiesRouteur) === 4) {
            $prefixe3 = "{$partiesRouteur[0]}.{$partiesRouteur[1]}.{$partiesRouteur[2]}.";
            foreach (ReseauService::toutes() as $ipLocale) {
                if (str_starts_with($ipLocale, $prefixe3)) {
                    return $ipLocale;
                }
            }
        }

        return null;
    }

        try {
            foreach ($this->queryFiltered('/ip/hotspot/host/print', ['mac-address' => $mac]) as $h) {
                if (isset($h['.id'])) $this->queryWithId('/ip/hotspot/host/remove', $h['.id']);
            }
        } catch (\Throwable) {
        }
    }
    
    // Helpers
    private function replyHasDone(array $reply): bool
    {
        foreach ($reply as $item) if ($item['type'] === '!done') return true;
        return false;
    }

    private function replyHasTrap(array $reply): bool
    {
        foreach ($reply as $item) if ($item['type'] === '!trap' || $item['type'] === '!fatal') return true;
        return false;
    }

    private function extractTrapMessage(array $reply): ?string
    {
        foreach ($reply as $item) {
            if ($item['type'] === '!trap' || $item['type'] === '!fatal') {
                return $item['attributes']['message'] ?? $item['attributes']['category'] ?? 'Erreur inconnue';
            }
        }
        return null;
    }

    /**
     * Ajoute une règle DNS statique sur le routeur pour associer le nom du portail
     * à l'adresse IP du serveur.
     */
    public function configurerDnsLocal(string $domaine, string $ip): void
    {
        $existantes = $this->queryFiltered('/ip/dns/static/print', ['name' => $domaine]);

        if (!empty($existantes)) {
            $id = $existantes[0]['.id'];
            $this->queryWithId('/ip/dns/static/set', $id, ['address' => $ip]);
        } else {
            $this->query('/ip/dns/static/add', [
                'name'    => $domaine,
                'address' => $ip,
                'comment' => 'Portail Captif (Auto)',
            ]);
        }
    }

    /** Vide le cache DNS du routeur, sans toucher au HotSpot (les clients restent connectés). */
    public function viderCacheDns(): void
    {
        try {
            $this->query('/ip/dns/cache/flush');
        } catch (\Throwable) {
        }
    }

namespace App\Services;

use Illuminate\Support\Facades\Cache;

/**
 * Statut de connexion MikroTik pour la barre supérieure.
 * Durée de cache courte (8 s) pour une réactivité immédiate sans ralentir les requêtes.
 */
class MikroTikStatusService
{
    private const CLE = 'mikrotik.statut';
    private const TTL = 8;

    public function __construct(
        private MikroTikService $mikroTik,
    ) {
    }

    /**
     * @return array{configure:bool, connecte:bool, identity:?string, version:?string, erreur:?string, verifie_a:string}
     */
    public function obtenir(): array
    {
        if (!$this->mikroTik->estConfigure()) {
            return [
                'configure' => false,
                'connecte'  => false,
                'identity'  => null,
                'version'   => null,
                'erreur'    => null,
                'verifie_a' => now()->toIso8601String(),
            ];
        }

        return Cache::remember(self::CLE, self::TTL, function () {
            try {
                $info = $this->mikroTik->testConnection();

                return [
                    'configure' => true,
                    'connecte'  => true,
                    'identity'  => $info['identity'],
                    'version'   => $info['version'],
                    'erreur'    => null,
                    'verifie_a' => now()->toIso8601String(),
                ];
            } catch (\Throwable $e) {
                return [
                    'configure' => true,
                    'connecte'  => false,
                    'identity'  => null,
                    'version'   => null,
                    'erreur'    => mb_substr($e->getMessage(), 0, 200),
                    'verifie_a' => now()->toIso8601String(),
                ];
            }
        });
    }
}

<?php

namespace App\Services\MikroTik;

use RuntimeException;

/**
 * Connexion TCP bas niveau à l'API RouterOS (port 8728 / 8729 SSL).
 *
 * Ne contient AUCUNE logique métier : uniquement l'encodage/décodage
 * du protocole "sentences" de RouterOS.
 *
 * Compatible RouterOS 6.43+ (login simple) et 7.x.
 */
class MikrotikConnection
{
    /** @var resource|null */
    private $socket = null;

    public function __construct(
        private string $host,
        private int $port = 8728,
        private float $timeout = 5.0,
        private bool $useSsl = false,
    ) {
    }

    public function connect(): void
    {
        if ($this->socket !== null) {
            return;
        }

        $transport = $this->useSsl ? 'ssl' : 'tcp';
        $address   = "{$transport}://{$this->host}:{$this->port}";

        // Pour SSL auto-signé (RouterOS par défaut). À durcir plus tard.
        $context = stream_context_create([
            'ssl' => [
                'verify_peer'      => false,
                'verify_peer_name' => false,
            ],
        ]);

        $errno  = 0;
        $errstr = '';

        $socket = @stream_socket_client(
            $address,
            $errno,
            $errstr,
            $this->timeout,
            STREAM_CLIENT_CONNECT,
            $context
        );

        if ($socket === false) {
            throw new RuntimeException(
                "Connexion MikroTik impossible ({$this->host}:{$this->port}) : "
                . "{$errstr} (errno {$errno})"
            );
        }

        stream_set_timeout($socket, (int) $this->timeout);
        $this->socket = $socket;
    }

    public function disconnect(): void
    {
        if ($this->socket !== null) {
            @fclose($this->socket);
            $this->socket = null;
        }
    }

    public function isConnected(): bool
    {
        return $this->socket !== null;
    }

    /**
     * Envoie une phrase (liste de mots) et retourne la réponse complète
     * jusqu'à !done ou !fatal.
     *
     * @param  string[]  $words
     * @return array<int, array{type: string, attributes: array<string,string>}>
     */
    public function sendSentence(array $words): array
    {
        if ($this->socket === null) {
            throw new RuntimeException('Non connecté au MikroTik.');
        }

        foreach ($words as $word) {
            $this->writeWord($word);
        }
        // Fin de phrase = mot vide
        $this->writeWord('');

        return $this->readReply();
    }

    // -----------------------------------------------------------------
    // Encodage / décodage
    // -----------------------------------------------------------------

    private function writeWord(string $word): void
    {
        fwrite($this->socket, $this->encodeLength(strlen($word)) . $word);
    }

    private function encodeLength(int $length): string
    {
        if ($length < 0x80) {
            return chr($length);
        }
        if ($length < 0x4000) {
            $length |= 0x8000;
            return chr(($length >> 8) & 0xFF) . chr($length & 0xFF);
        }
        if ($length < 0x200000) {
            $length |= 0xC00000;
            return chr(($length >> 16) & 0xFF)
                 . chr(($length >> 8) & 0xFF)
                 . chr($length & 0xFF);
        }
        if ($length < 0x10000000) {
            $length |= 0xE0000000;
            return chr(($length >> 24) & 0xFF)
                 . chr(($length >> 16) & 0xFF)
                 . chr(($length >> 8) & 0xFF)
                 . chr($length & 0xFF);
        }
        return chr(0xF0)
             . chr(($length >> 24) & 0xFF)
             . chr(($length >> 16) & 0xFF)
             . chr(($length >> 8) & 0xFF)
             . chr($length & 0xFF);
    }

    private function readLength(): int
    {
        $byte = ord($this->readBytes(1));

        if (($byte & 0x80) === 0x00) {
            return $byte;
        }
        if (($byte & 0xC0) === 0x80) {
            $b2 = ord($this->readBytes(1));
            return (($byte & 0x3F) << 8) | $b2;
        }
        if (($byte & 0xE0) === 0xC0) {
            $rest = $this->readBytes(2);
            return (($byte & 0x1F) << 16) | unpack('n', $rest)[1];
        }
        if (($byte & 0xF0) === 0xE0) {
            $rest = $this->readBytes(3);
            return (($byte & 0x0F) << 24) | unpack('N', "\x00" . $rest)[1];
        }
        if (($byte & 0xF8) === 0xF0) {
            return unpack('N', $this->readBytes(4))[1];
        }

        throw new RuntimeException(
            'Longueur MikroTik invalide : 0x' . dechex($byte)
        );
    }

    private function readBytes(int $n): string
    {
        $buffer = '';
        while (strlen($buffer) < $n) {
            $chunk = fread($this->socket, $n - strlen($buffer));

            if ($chunk === false || $chunk === '') {
                $info = stream_get_meta_data($this->socket);
                if (!empty($info['timed_out'])) {
                    throw new RuntimeException('Timeout de lecture MikroTik.');
                }
                if (feof($this->socket)) {
                    throw new RuntimeException('Connexion MikroTik fermée.');
                }
                throw new RuntimeException('Lecture MikroTik échouée.');
            }
            $buffer .= $chunk;
        }
        return $buffer;
    }

    /**
     * Lit une réponse complète jusqu'à !done ou !fatal.
     */
    private function readReply(): array
    {
        $reply = [];

        while (true) {
            $sentence = $this->readSentence();

            if (empty($sentence)) {
                continue;
            }

            $type       = array_shift($sentence);
            $attributes = [];

            foreach ($sentence as $word) {
                if ($word === '' || $word[0] !== '=') {
                    continue;
                }
                $rest  = substr($word, 1);
                $eqPos = strpos($rest, '=');
                if ($eqPos === false) {
                    $attributes[$rest] = '';
                } else {
                    $attributes[substr($rest, 0, $eqPos)] = substr($rest, $eqPos + 1);
                }
            }

            $reply[] = [
                'type'       => $type,
                'attributes' => $attributes,
            ];

            if ($type === '!done' || $type === '!fatal') {
                return $reply;
            }
        }
    }

    /**
     * @return string[]
     */
    private function readSentence(): array
    {
        $words = [];
        while (true) {
            $length = $this->readLength();
            if ($length === 0) {
                return $words;
            }
            $words[] = $this->readBytes($length);
        }
    }
}
