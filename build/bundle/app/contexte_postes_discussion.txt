<script>
const POSTES_URL = @json(url('/postes'));
const TARIFS_RACCOURCIS = @json($tarifs);
const TARIF_MINUTE = {{ $tarif_minute ?? 20 }};
const TARIF_MIN = {{ $tarif_min ?? 300 }};
const TARIF_ARRONDI = @json($tarif_arrondi ?? ['actif' => false, 'unite' => 0, 'seuil' => 0]);

const COULEURS = { disponible: '#22c55e', occupe: '#f59e0b', suspendu: '#8b5cf6', verrouille: '#ef4444', desactive: '#94a3b8', hors_ligne: '#94a3b8' };
const LIBELLES = { disponible: 'Disponible', occupe: 'Utilisé', suspendu: 'En pause', verrouille: 'Verrouillé', desactive: 'Désactivé', hors_ligne: 'Non présent' };
const INTERNET = { autorise: 'Autorisé', coupe: 'Coupé', erreur: 'Erreur routeur', non_gere: 'Non contrôlé (routeur non configuré)' };

function svgIcone(etat, taille = 'h-9 w-9') {
    const c = COULEURS[etat] || '#94a3b8';
    let badge = '';
    if (etat === 'hors_ligne') {
        badge = '<circle cx="31" cy="31" r="8" fill="#dc2626" stroke="#fff" stroke-width="1.5"/><rect x="30" y="26" width="2" height="6" fill="#fff"/><rect x="30" y="33" width="2" height="2" fill="#fff"/>';
    } else if (etat === 'desactive') {
        badge = '<line x1="6" y1="30" x2="30" y2="6" stroke="#475569" stroke-width="3" stroke-linecap="round"/>';
    } else if (etat === 'verrouille') {
        badge = '<rect x="26" y="25" width="10" height="8" rx="1.5" fill="#fff" stroke="#b91c1c" stroke-width="1.5"/><path d="M28 25v-3a3 3 0 0 1 6 0v3" fill="none" stroke="#b91c1c" stroke-width="1.5"/>';
    } else if (etat === 'suspendu') {
        badge = '<rect x="12" y="10" width="3.5" height="11" fill="#fff"/><rect x="20.5" y="10" width="3.5" height="11" fill="#fff"/>';
    }
    return `<svg viewBox="0 0 38 38" class="${taille}"><rect x="3" y="4" width="26" height="19" rx="2.5" fill="${c}" stroke="#1e293b" stroke-width="1.5"/><rect x="6" y="7" width="20" height="13" rx="1" fill="#0f172a" opacity=".35"/><rect x="13" y="23" width="6" height="4" fill="#64748b"/><rect x="9" y="27" width="14" height="2.5" rx="1" fill="#475569"/>${badge}</svg>`;
}

function fmtDuree(s) {
    if (s === null || s === undefined) return '—';
    s = Math.max(0, Math.floor(s));
    const p = n => String(n).padStart(2, '0');
    return `${p(Math.floor(s / 3600))}:${p(Math.floor((s % 3600) / 60))}:${p(s % 60)}`;
}
const fmtAr = n => (Number(n) || 0).toLocaleString('fr-FR') + ' Ar';

function montantPourMinutes(minutes) {
    return Math.max(TARIF_MIN, Math.max(0, Math.floor(minutes)) * TARIF_MINUTE);
}

const BOUTONS = { vert: 'bg-emerald-600 hover:bg-emerald-700 text-white', rouge: 'bg-rose-600 hover:bg-rose-700 text-white', ambre: 'bg-amber-500 hover:bg-amber-600 text-white', violet: 'bg-violet-600 hover:bg-violet-700 text-white', bleu: 'bg-sky-600 hover:bg-sky-700 text-white', gris: 'bg-slate-600 hover:bg-slate-700 text-white' };

function postesApp() {
    return {
        postes: @json($postes),
        selId: null,
        menu: { ouvert: false, x: 0, y: 0, id: null },
        modal: null,
        confirm: { ouvert: false, action: null, id: null, titre: '', texte: '', callback: null },
        form: { montantManuel: '', h: 0, m: 0 },
        lignesForfait: [...TARIFS_RACCOURCIS],
        ligneSelectionnee: null,
        newLigne: { type: 'montant', montant: '', h: 0, m: 0 },
        erreurLigne: '',
        enregLigne: false,
        hs: { username: '', password: '', voir: false, erreur: '' },
        chat: [], chatDernier: 0, chatSession: null, texte: '', envoiChat: false,
        nonLusVus: {},
        maintenant: Date.now(), recuLe: Date.now(),
        horsServeur: false, notif: null, enCours: null,

        init() {
            this.postes.forEach(p => this.nonLusVus[p.id] = p.non_lus ?? 0);
            if (this.postes.length > 0 && !this.selId) {
                this.choisir(this.postes[0].id);
            }
            setInterval(() => this.maintenant = Date.now(), 1000);
            setInterval(() => this.rafraichir(), 3000);
            document.addEventListener('click', () => this.menu.ouvert = false);
            document.addEventListener('keydown', e => {
                if (e.key === 'Escape') {
                    if (this.confirm.ouvert) { this.confirm.ouvert = false; return; }
                    this.menu.ouvert = false; this.modal = null;
                }
            });
        },

        get sel() { return this.postes.find(p => p.id === this.selId) || null; },
        get bilan() {
            const n = e => this.postes.filter(p => p.etat === e).length;
            return `${this.postes.length} poste(s) — ${n('occupe') + n('suspendu') + n('verrouille')} utilisé(s), ${n('hors_ligne')} non présent(s)`;
        },
        get totalNonLus() { return this.postes.filter(p => p.id !== this.selId).reduce((t, p) => t + (p.non_lus || 0), 0); },

        icone: svgIcone, libelle(e) { return LIBELLES[e] || e; }, mode(m) { return (m === 'forfait' ? 'Forfait' : (m === 'postpaye' ? 'Postpayé' : (m === 'libre' ? 'Session libre' : m))); },
        internet(v) { return INTERNET[v] || 'En attente'; },
        classeBouton(c) { return BOUTONS[c] || BOUTONS.gris; }, fmtAr,

        decalage(p) { if (!p.session || !['occupe', 'verrouille'].includes(p.etat)) return 0; return Math.floor((this.maintenant - this.recuLe) / 1000); },
        ecoule(p) { return p.session ? fmtDuree(p.session.ecoule_s + this.decalage(p)) : '—'; },
        restant(p) { if (!p.session || p.session.restant_s === null) return '—'; return fmtDuree(p.session.restant_s - this.decalage(p)); },
        tempsListe(p) { if (!p.session) return ''; return p.session.restant_s !== null ? this.restant(p) : this.ecoule(p); },

        montantActuel(p) {
            if (!p.session) return '—';
            if (p.session.mode === 'libre') return 'Sans facturation';
            if (p.session.etat === 'terminee') return fmtAr(p.session.montant);

            let sec = p.session.ecoule_s + this.decalage(p);
            if (p.session.mode === 'forfait' && p.session.duree_min) sec = Math.min(sec, p.session.duree_min * 60);
            return fmtAr(montantPourMinutes(Math.floor(sec / 60)));
        },

        debutAffiche(p) {
            const s = p?.session;
            if (!s) return '—';
            if (s.debut_iso) {
                const d = new Date(s.debut_iso);
                if (!isNaN(d.getTime())) return d.toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit', second: '2-digit' });
            }
            return s.debut || '—';
        },

        dureeMinimale() { return TARIF_MINUTE > 0 ? Math.ceil(TARIF_MIN / TARIF_MINUTE) : 0; },

        async rafraichir() {
            try {
                const r = await cmFetch(POSTES_URL + '/etat');
                const d = await r.json();
                for (const p of d.postes) {
                    const avant = this.nonLusVus[p.id] ?? 0;
                    if ((p.non_lus || 0) > avant && p.id !== this.selId) this.afficher(`Nouveau message de ${p.nom}`, true);
                    this.nonLusVus[p.id] = p.non_lus || 0;
                }
                this.postes = d.postes;
                this.recuLe = Date.now();
                this.horsServeur = false;
                if (this.selId && !this.sel) this.selId = null;

                if (this.selId) {
                    const sidNow = this.sel?.session?.id ?? null;
                    if (sidNow !== this.chatSession) {
                        this.chatSession = sidNow;
                        this.chat = []; this.chatDernier = 0;
                        if (sidNow) await this.chargerChat(true);
                    }
                }
                if (this.selId && (this.sel?.etat === 'occupe' || this.sel?.etat === 'desactive')) this.chargerChat(false);
            } catch (e) { this.horsServeur = true; }
        },

        choisir(id) {
            this.selId = id; this.chat = []; this.chatDernier = 0;
            this.chatSession = this.sel?.session?.id ?? null;
            if (this.sel?.session) this.chargerChat(true);
        },

        ouvrirMenu(e, id) {
            if (this.selId !== id) this.choisir(id);
            this.menu = { ouvert: true, x: Math.min(e.clientX, window.innerWidth - 220), y: Math.min(e.clientY, window.innerHeight - 320), id };
        },

        demanderConfirm(id, action) {
            const def = (this.postes.find(p => p.id === id)?.actions || []).find(a => a.cle === action);
            if (!def) return;
            this.menu.ouvert = false;
            if (this.selId !== id) this.choisir(id);
            if (def.modal) {
                this.form = { montantManuel: '', h: 0, m: 0 };
                this.ligneSelectionnee = null;
                this.erreurLigne = '';
                this.modal = action;
                return;
            }
            if (def.confirm) {
                this.confirm = { ouvert: true, action, id, titre: def.label, texte: def.confirm, callback: null };
                return;
            }
            this.executer(id, action, {});
        },

        async confirmerAction() {
            const { id, action } = this.confirm;
            this.confirm.ouvert = false;
            await this.executer(id, action, {});
        },

        async executer(id, action, corps) {
            if (this.enCours) return;
            this.enCours = action;
            try {
                const r = await cmFetch(`${POSTES_URL}/${id}/${action}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(corps) });
                const d = await r.json().catch(() => ({}));
                this.afficher(d.message || (r.ok ? 'Action effectuée.' : 'Erreur.'), r.ok);
                if (r.ok) this.modal = null;
            } catch (e) { this.afficher('Serveur injoignable.', false); }
            this.enCours = null;
            await this.rafraichir();
        },

        async ajouterLigne() {
            this.erreurLigne = '';
            let m = 0;
            if (this.newLigne.type === 'montant') m = Math.round(Number(this.newLigne.montant));
            else m = (((Number(this.newLigne.h) || 0) * 60) + (Number(this.newLigne.m) || 0)) * TARIF_MINUTE;

            if (!m || m <= 0) { this.erreurLigne = 'Saisissez un montant ou une durée valide.'; return; }
            if (m < TARIF_MIN) { this.erreurLigne = `Montant trop faible : le minimum est de ${fmtAr(TARIF_MIN)}.`; return; }

            const duree = Math.round(m / TARIF_MINUTE);
            if (this.lignesForfait.some(l => Number(l.montant) === m || Number(l.duree) === duree)) {
                this.erreurLigne = 'Une recharge identique (même montant ou même durée) existe déjà dans le tableau.';
                return;
            }
            if (this.enregLigne) return;
            this.enregLigne = true;

            try {
                const r = await cmFetch(POSTES_URL + '/raccourcis', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' },
                    body: JSON.stringify({ montant: m })
                });
                const d = await r.json().catch(() => ({}));
                if (!r.ok || !d.ok) {
                    this.erreurLigne = d.message || 'Enregistrement impossible.';
                } else {
                    this.lignesForfait.push({ id: d.id, montant: d.montant, duree: d.duree });
                    this.lignesForfait.sort((a, b) => a.montant - b.montant);
                    this.newLigne = { type: this.newLigne.type, montant: '', h: 0, m: 0 };
                }
            } catch (e) { this.erreurLigne = 'Serveur injoignable.'; }
            this.enregLigne = false;
        },

        demanderSupprimerLigne(index) {
            const ligne = this.lignesForfait[index];
            this.confirm = {
                ouvert: true,
                action: 'supprimer_ligne',
                id: null,
                titre: 'Supprimer',
                texte: 'Supprimer cette recharge du tableau ?',
                callback: async () => {
                    this.confirm.ouvert = false;
                    if (!ligne) return;
                    if (ligne.id) {
                        try {
                            const r = await cmFetch(`${POSTES_URL}/raccourcis/${ligne.id}/supprimer`, {
                                method: 'POST',
                                headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' },
                                body: '{}'
                            });
                            const d = await r.json().catch(() => ({}));
                            if (!r.ok || !d.ok) { this.afficher(d.message || 'Suppression impossible.', false); return; }
                        } catch (e) { this.afficher('Serveur injoignable.', false); return; }
                    }
                    this.lignesForfait = this.lignesForfait.filter(l => Number(l.montant) !== Number(ligne.montant));
                }
            };
        },

        selectionnerLigne(l) {
            this.ligneSelectionnee = l.montant;
            this.form.montantManuel = '';
            this.form.h = 0; this.form.m = 0;
        },

        calculerDepuisMontant() {
            this.ligneSelectionnee = null;
            const m = Number(this.form.montantManuel);
            if (m > 0) {
                const totalMin = m / TARIF_MINUTE;
                this.form.h = Math.floor(totalMin / 60);
                this.form.m = Math.round(totalMin % 60);
            }
        },

        calculerDepuisDuree() {
            this.ligneSelectionnee = null;
            const d = (Number(this.form.h) * 60) + Number(this.form.m);
            if (d > 0) this.form.montantManuel = d * TARIF_MINUTE;
        },

        validerModal() {
            const id = this.selId;
            let montant = 0;
            if (this.ligneSelectionnee) montant = this.ligneSelectionnee;
            else if (this.form.montantManuel) montant = Number(this.form.montantManuel);
            else if (this.form.h || this.form.m) montant = ((Number(this.form.h) * 60) + Number(this.form.m)) * TARIF_MINUTE;

            if (montant <= 0) { this.afficher("Veuillez sélectionner ou saisir un montant.", false); return; }
            const duree = Math.round(montant / TARIF_MINUTE);

            if (this.modal === 'demarrer') {
                const dMin = this.dureeMinimale();
                if (montant < TARIF_MIN || duree < dMin) {
                    this.afficher(`Première activation : minimum ${fmtAr(TARIF_MIN)} (${dMin} min).`, false, 4500);
                    return;
                }
            }

            if (this.modal === 'demarrer') this.executer(id, 'demarrer', { type: 'forfait', duree_minutes: duree });
            else this.executer(id, 'recharger', { minutes: duree });
        },

        async chargerChat(reset) {
            const id = this.selId;
            if (!id) return;
            try {
                const r = await cmFetch(`${POSTES_URL}/${id}/chat?apres=${reset ? 0 : this.chatDernier}`);
                const d = await r.json();
                if (id !== this.selId) return;
                if (reset) this.chat = [];
                if (d.messages?.length) {
                    const existants = new Set(this.chat.map(m => m.id));
                    d.messages.forEach(m => { if (!existants.has(m.id)) { this.chat.push(m); existants.add(m.id); } });
                    this.chat.sort((a, b) => a.id - b.id);
                    this.chatDernier = d.messages[d.messages.length - 1].id;
                    this.$nextTick(() => { const b = this.$refs.chatBox; if (b) b.scrollTop = b.scrollHeight; });
                }
            } catch (e) {}
        },

        async envoyerMessage() {
            const t = this.texte.trim();
            if (!t || !this.selId || this.envoiChat) return;
            const id = this.selId;
            this.envoiChat = true;
            try {
                const r = await cmFetch(`${POSTES_URL}/${id}/message`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message: t }) });
                const d = await r.json().catch(() => ({}));
                if (r.ok && d.ok) {
                    this.texte = '';
                    if (id === this.selId && d.id && !this.chat.some(m => m.id === d.id)) {
                        this.chat.push({ id: d.id, sens: 'admin', contenu: t, heure: d.heure || new Date().toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' }) });
                        this.chat.sort((a, b) => a.id - b.id);
                    }
                    this.$nextTick(() => { const b = this.$refs.chatBox; if (b) b.scrollTop = b.scrollHeight; });
                } else {
                    this.afficher(d.message || 'Message non envoyé.', false);
                }
            } catch (e) { this.afficher('Serveur injoignable.', false); }
            this.envoiChat = false;
        },

        afficher(msg, ok, ms = 3000) { this.notif = { msg, ok }; setTimeout(() => this.notif = null, ms); },
    };
}
</script>

    public function __construct(private PosteSessionService $sessions, private AccesInternetService $internet) {}

    public function index()
    {
        $tarif = app(TarifService::class)->tarifStandard();

        return view('postes.index', [
            'postes' => $this->tousLesPostes(),
            'tarifs' => $tarif->raccourcis()->get(),
            'tarif_minute' => $tarif->montant_par_minute,
            'tarif_min' => $tarif->montant_minimum,
            'tarif_arrondi' => [

    public function etat(): JsonResponse
    {
        $this->internet->reconcilierTous();
        return response()->json(['ok' => true, 'postes' => $this->tousLesPostes()]);
    }

    public function chat(Request $request, Poste $poste): JsonResponse
    {
        $apres = (int) $request->query('apres', 0);
        $messages = $poste->messages()->where('id', '>', $apres)->orderBy('id')->limit(100)->get(['id', 'sens', 'contenu', 'created_at', 'lu_at']);

        $nonLus = $messages->where('sens', PosteMessage::SENS_CLIENT)->whereNull('lu_at')->pluck('id');
        if ($nonLus->isNotEmpty()) PosteMessage::whereIn('id', $nonLus)->update(['lu_at' => now()]);

        return response()->json([
            'ok'       => true,
            'messages' => $messages->map(fn ($m) => ['id' => $m->id, 'sens' => $m->sens, 'contenu' => $m->contenu, 'heure' => $m->created_at?->format('H:i')])->values(),
        ]);
    }

    public function demarrer(Request $request, Poste $poste)
    {
        $data = $request->validate([
            'type'          => ['required', 'in:forfait,postpaye'],
            'duree_minutes' => ['required_if:type,forfait', 'nullable', 'integer', 'min:1', 'max:1440'],
        ]);

        if ($poste->etatAffiche() !== Poste::ETAT_DISPONIBLE && $poste->etatAffiche() !== Poste::ETAT_DESACTIVE) {
            return $this->echec($request, "Le poste n'est pas disponible.");
        }

        if ($data['type'] === 'forfait') {
            try {
                $tarif   = app(TarifService::class)->tarifStandard();
                $pm      = max(1, (int) $tarif->montant_par_minute);
                $minimum = (int) $tarif->montant_minimum;
                $dureeMin = (int) ceil($minimum / $pm);

                if ($minimum > 0 && (int) $data['duree_minutes'] < $dureeMin) {
                    return $this->echec($request, "Première activation : minimum {$minimum} Ar ({$dureeMin} min).");
                }
            } catch (\Throwable) {
            }
        }

        $duree = $data['type'] === 'forfait' ? (int) $data['duree_minutes'] : null;
        $session = $this->sessions->creerSession($poste, $data['type'], $duree);
        $reseau  = $this->internet->autoriser($poste);

        PosteMessage::where('poste_id', $poste->id)->delete();

        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'demarrer', 'payload' => ['session_id' => $session->id, 'duree_minutes' => $duree, 'type' => $data['type']]]);
        $poste->update(['etat' => Poste::ETAT_OCCUPE]);

        return $this->ok($request, "Poste {$poste->nom_poste} démarré." . $this->avertissementReseau($reseau));
    }

    public function debloquer(Request $request, Poste $poste)
    {
        if (!$poste->sessionActive()) $this->sessions->creerSession($poste, 'libre', null);
        $this->sessions->reprendre($poste);
        $reseau = $this->internet->autoriser($poste);
        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'debloquer']);
        return $this->ok($request, "Poste {$poste->nom_poste} débloqué." . $this->avertissementReseau($reseau));
    }

    public function fermer(Request $request, Poste $poste)
    {
        $session = $this->terminer($poste, 'admin');
        if ($session === null) return $this->echec($request, 'Aucune session active à fermer.');

        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'verrouiller']);
        $poste->update(['etat' => Poste::ETAT_DESACTIVE]);

        $montant = ($session->mode_facturation ?? 'forfait') === 'libre' ? '' : ' Montant : ' . number_format((int) $session->montant_total, 0, ',', ' ') . ' Ar.';
        return $this->ok($request, "Session de {$poste->nom_poste} fermée.{$montant}");
    }

    public function suspendre(Request $request, Poste $poste)
    {
        $ok = DB::transaction(function () use ($poste) {
            $p = Poste::whereKey($poste->id)->lockForUpdate()->first();
            return $p !== null && $this->sessions->suspendre($p);
        });

        if (!$ok) return $this->echec($request, 'Aucune session active.');
        $reseau = $this->internet->couper($poste);
        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'suspendre']);
        return $this->ok($request, "Session suspendue." . $this->avertissementReseau($reseau, 'coupé'));
    }

    public function reprendre(Request $request, Poste $poste)
    {
        if (!$this->sessions->reprendre($poste)) return $this->echec($request, 'Aucune session à reprendre.');
        $reseau = $this->internet->autoriser($poste);
        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'reprendre']);
        return $this->ok($request, "Session reprise." . $this->avertissementReseau($reseau));
    }

    public function verrouiller(Request $request, Poste $poste)
    {
        if (!$poste->sessionActive()) return $this->echec($request, 'Aucune session active.');
        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'verrouiller']);
        $poste->update(['etat' => Poste::ETAT_VERROUILLE]);
        return $this->ok($request, "Poste verrouillé.");
    }

    public function desactiver(Request $request, Poste $poste)
    {
        if ($poste->etatAffiche() === Poste::ETAT_OCCUPE) return $this->echec($request, "Impossible de désactiver un poste en cours d'utilisation.");
        $this->terminer($poste, 'desactivation');
        PosteMessage::where('poste_id', $poste->id)->delete();
        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'desactiver']);
        $poste->update(['etat' => Poste::ETAT_DESACTIVE]);
        return $this->ok($request, "Poste désactivé.");
    }

    public function activer(Request $request, Poste $poste)
    {
        if ($poste->etat === Poste::ETAT_DESACTIVE) {
            $poste->update(['etat' => Poste::ETAT_DISPONIBLE]);
            PosteMessage::where('poste_id', $poste->id)->delete();
            PosteCommande::create(['poste_id' => $poste->id, 'type' => 'verrouiller']);
            return $this->ok($request, "Poste activé.");
        }
        return $this->echec($request, "Le poste n'est pas désactivé.");
    }

    public function arreter(Request $request, Poste $poste)
    {
        $this->terminer($poste, 'arret');
        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'arreter']);
        $poste->update(['etat' => Poste::ETAT_DESACTIVE]);
        return $this->ok($request, "Commande d'arrêt envoyée.");
    }

    public function recharger(Request $request, Poste $poste)
    {
        $data = $request->validate(['minutes' => ['required', 'integer', 'min:1', 'max:1440']]);

        $resultat = DB::transaction(function () use ($poste, $data) {
            $p       = Poste::whereKey($poste->id)->lockForUpdate()->first();
            $session = $p?->sessionActive();

            if ($session === null) return 'aucune';
            if (($session->mode_facturation ?? 'forfait') !== 'forfait') return 'non_forfait';

            $nouvelleDuree = (int) $session->duree_prevue + $data['minutes'];
            $session->update([
                'date_heure_fin_prevue' => $session->date_heure_fin_prevue->copy()->addMinutes($data['minutes']),
                'duree_prevue' => $nouvelleDuree,
            ]);
            return $nouvelleDuree;
        });

        if ($resultat === 'aucune') return $this->echec($request, 'Aucune session active.');
        if ($resultat === 'non_forfait') return $this->echec($request, "Seul un forfait peut être rechargé.");

        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'recharger', 'payload' => ['minutes_ajoutees' => (int) $data['minutes']]]);
        return $this->ok($request, "{$data['minutes']} min ajoutées (forfait : {$resultat} min).");
    }

    public function envoyerMessage(Request $request, Poste $poste)
    {
        $data = $request->validate(['message' => ['required', 'string', 'max:500']]);
        $m = PosteMessage::create(['poste_id' => $poste->id, 'sens' => PosteMessage::SENS_ADMIN, 'contenu' => $data['message']]);
        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'message', 'payload' => ['message' => $data['message']]]);

        if ($request->expectsJson()) {
            return response()->json([
                'ok'      => true,
                'message' => 'Message envoyé.',
                'id'      => $m->id,
                'heure'   => $m->created_at?->format('H:i'),
            ]);
        }
        return back()->with('success', 'Message envoyé.');
    }

    public function ajouterRaccourci(Request $request): JsonResponse
    {
        $data = $request->validate(['montant' => ['required', 'integer', 'min:1', 'max:10000000']]);

        try {
            $tarif = app(TarifService::class)->tarifStandard();
        } catch (\Throwable) {
            return response()->json(['ok' => false, 'message' => 'Aucun tarif configuré.'], 422);
        }

        $montant = (int) $data['montant'];
        $pm      = (int) $tarif->montant_par_minute;

        if ($pm <= 0) return response()->json(['ok' => false, 'message' => 'Prix par minute invalide.'], 422);
        if ($tarif->montant_minimum > 0 && $montant < $tarif->montant_minimum) {
            return response()->json(['ok' => false, 'message' => "Montant trop faible : le minimum est de {$tarif->montant_minimum} Ar."], 422);
        }

        $duree = (int) round($montant / $pm);
        if ($duree < 1) return response()->json(['ok' => false, 'message' => 'Durée invalide.'], 422);

        $existe = $tarif->raccourcis()
            ->where(fn ($q) => $q->where('montant', $montant)->orWhere('duree', $duree))
            ->exists();
        if ($existe) {
            return response()->json(['ok' => false, 'message' => 'Une recharge identique (même montant ou même durée) existe déjà dans le tableau.'], 422);
        }

        $r = new TarifRaccourci();
        $r->forceFill(['tarif_id' => $tarif->id, 'montant' => $montant, 'duree' => $duree, 'ordre' => 0])->save();

        return response()->json(['ok' => true, 'id' => $r->id, 'montant' => $montant, 'duree' => $duree]);
    }

    public function supprimerRaccourci(Request $request, int $id): JsonResponse
    {
        $r = TarifRaccourci::find($id);
        if ($r === null) return response()->json(['ok' => true]);

        $r->delete();
        return response()->json(['ok' => true]);
    }

    private function tousLesPostes(): array
    {
        return Poste::orderBy('nom_poste')
            ->withCount(['messages as non_lus' => fn ($q) => $q->where('sens', PosteMessage::SENS_CLIENT)->whereNull('lu_at')])
            ->get()->map(fn (Poste $p) => $this->donnees($p))->values()->all();
    }

    private function donnees(Poste $p): array
    {
        $etat = $p->etatAffiche();
        $s    = $p->sessionActive();
        if (!$s && $etat === Poste::ETAT_DESACTIVE) $s = $p->sessions()->latest()->first();

        $session = null;
        $estChatActif = $etat === Poste::ETAT_OCCUPE || ($etat === Poste::ETAT_DESACTIVE && $s !== null);

        if ($s) {
            $mode    = $s->mode_facturation ?: 'forfait';
            $ecoule  = (int) $s->secondesConsommees();
            $prevue  = (int) $s->duree_prevue;
            $restant = ($mode === 'forfait' && $prevue > 0) ? max(0, $prevue * 60 - $ecoule) : null;

            $forfaitTotal = 0;
            if ($mode === 'forfait') {
                $tarif = app(\App\Services\TarifService::class)->tarifStandard();
                $forfaitTotal = app(\App\Services\TarifService::class)->montantPourMinutes($prevue, $tarif);
            }

            $session = [
                'id'         => $s->id,
                'mode'       => $mode,
                'etat'       => $s->etat,
                'debut'      => $s->date_heure_debut?->format('H:i:s'),
                'debut_iso'  => $s->date_heure_debut?->toIso8601String(),
                'ecoule_s'   => $ecoule,
                'restant_s'  => $restant,
                'duree_min'  => $prevue ?: null,
                'forfait_total' => $forfaitTotal,
                'montant'    => $mode === 'libre' ? 0 : ($s->etat === 'terminee' ? (int) $s->montant_total : $this->sessions->montant($s, $ecoule)),
                'pause'      => SessionTransparence::pour($s),   // ← AJOUT
            ];
        }

        return [
            'id'       => $p->id,
            'nom'      => $p->nom_poste,
            'etat'     => $etat,
            'ip'       => $p->adresse_ip,
            'mac'      => $p->adresse_mac,
            'version'  => $p->agent_version,
            'session'  => $session,
            'non_lus'  => $estChatActif ? (int) ($p->non_lus ?? 0) : 0,
            'internet' => $p->internet_etat,
            'actions'  => $this->actionsPour($etat, $s !== null, $session['mode'] ?? null),
        ];
    }

    private function actionsPour(string $etat, bool $aSession, ?string $modeSession = null): array
    {
        $cles = match ($etat) {
            Poste::ETAT_DISPONIBLE => ['demarrer', 'debloquer', 'desactiver', 'arreter'],
            Poste::ETAT_OCCUPE     => ['fermer', 'suspendre', 'verrouiller', 'recharger', 'arreter'],
            Poste::ETAT_SUSPENDU   => ['reprendre', 'fermer', 'desactiver', 'arreter'],
            Poste::ETAT_VERROUILLE => $aSession ? ['debloquer', 'fermer', 'recharger', 'desactiver', 'arreter'] : ['debloquer', 'desactiver', 'arreter'],
            Poste::ETAT_HORS_LIGNE => ['desactiver'],
            Poste::ETAT_DESACTIVE  => ['activer'],
            default                => [],
        };

        if ($modeSession !== 'forfait') $cles = array_values(array_diff($cles, ['recharger']));
        return array_map(fn (string $cle) => ['cle' => $cle] + self::ACTIONS[$cle], $cles);
    }

    private function terminer(Poste $poste, string $motif): ?Session
    {
        $arret = $poste->estEnLigne() ? null : $poste->dernier_heartbeat;
        $session = DB::transaction(function () use ($poste, $motif, $arret) {
            $p = Poste::whereKey($poste->id)->lockForUpdate()->first();
            return $p !== null ? $this->sessions->cloturer($p, $motif, $arret) : null;
        });
        $this->internet->couper($poste);
        return $session;
    }

    private function avertissementReseau(?bool $reseau, string $verbe = 'activé'): string
    {
        return $reseau === false ? " Attention : Internet n'a pas pu être {$verbe}." : '';
    }

    private function ok(Request $request, string $message)
    {
        return $request->expectsJson() ? response()->json(['ok' => true, 'message' => $message]) : back()->with('success', $message);
    }

    private function echec(Request $request, string $message)
    {
        return $request->expectsJson() ? response()->json(['ok' => false, 'message' => $message], 422) : back()->with('error', $message);
    }
}

    public const ETAT_OCCUPE      = 'occupe';
    public const ETAT_SUSPENDU    = 'suspendu';
    public const ETAT_VERROUILLE  = 'verrouille';
    public const ETAT_DESACTIVE   = 'desactive';
    public const ETAT_HORS_LIGNE  = 'hors_ligne';

    /** Délai (secondes) sans heartbeat avant de considérer le poste hors ligne. */
    public const DELAI_HORS_LIGNE = 180;

     * Deux signes de vie venant de deux adresses réseau différentes à moins de ce délai (secondes)
     * = deux ordinateurs utilisent la même identité (PC cloné).

    public const DELAI_GRACE_AGENT = 180;

            ->first();
    }

            && $this->dernier_heartbeat->gt(now()->subSeconds(self::delaiHorsLigne()));

        return $this->estEnLigne()
            && $this->etat !== self::ETAT_HORS_LIGNE
            && $this->etat !== self::ETAT_DESACTIVE;
    }

    public function etatAffiche(): string
    {
        if ($this->etat === self::ETAT_DESACTIVE) {
            return self::ETAT_DESACTIVE;
        }
        if (!$this->estEnLigne()) {
            return self::ETAT_HORS_LIGNE;
        }
        if ($this->etat === self::ETAT_HORS_LIGNE) {
            return $this->sessionActive() ? self::ETAT_OCCUPE : self::ETAT_DISPONIBLE;
        }
        return $this->etat;
    }

    /** Un conflit d'identité (PC cloné) est-il signalé pour ce poste ? */

    {
        return $this->conflit_identite_at !== null;

    // Noms de postes
    // ------------------------------------------------------------------

    /** Nettoie un nom : pas de caractères de contrôle, espaces simplifiés, 100 caractères maximum. */
    public static function nettoyerNom(string $nom): string
    {
        $nom = preg_replace('/[\x00-\x1F\x7F]+/u', ' ', $nom) ?? '';
        $nom = preg_replace('/\s+/u', ' ', $nom) ?? '';

        return mb_substr(trim($nom), 0, 100);
    }

    /** Ce nom (majuscules/minuscules ignorées) est-il déjà pris par un autre poste ? */
    public static function nomExiste(string $nom, ?int $ignorerId = null): bool
    {
        return static::query()
            ->whereRaw('LOWER(nom_poste) = ?', [mb_strtolower($nom)])
            ->when($ignorerId !== null, fn ($q) => $q->where('id', '!=', $ignorerId))
            ->exists();
    }

    /** Si le nom est pris, ajoute un suffixe : « Poste 3 » devient « Poste 3-2 », « Poste 3-3 »… */
    public static function nomDisponible(string $nom, ?int $ignorerId = null): string
    {
        $candidat = $nom;
        $i = 2;

        while (static::nomExiste($candidat, $ignorerId)) {
            $suffixe  = '-' . $i;
            $candidat = mb_substr($nom, 0, 100 - mb_strlen($suffixe)) . $suffixe;
            $i++;

            if ($i > 500) {   // garde-fou : ne jamais boucler indéfiniment
                $candidat = mb_substr($nom, 0, 90) . '-' . Str::lower(Str::random(6));
                break;
            }
        }

        return $candidat;
    }

    /**
     * Nom attribué à un NOUVEAU poste :
     *  - nom vide ou nom Windows par défaut (DESKTOP-XXXXXXX…) → « Poste 1 », « Poste 2 »…
     *  - sinon le nom demandé, avec un suffixe s'il est déjà pris.
     */
    public static function nomPourNouveauPoste(string $demande): string
    {
        $nom = self::nettoyerNom($demande);

        if ($nom === '' || preg_match('/^(DESKTOP|LAPTOP|WIN)-[A-Z0-9]{5,}$/i', $nom)) {
            $n = 1;
            while (static::nomExiste("Poste {$n}")) {
                $n++;
            }

            return "Poste {$n}";
        }

        return self::nomDisponible($nom);
    }

    public function creerSession(Poste $poste, int $dureeMinutes, string $type): Session
    {
        $tarif = Tarif::where('cible', 'ethernet')->orWhere('cible', 'unifie')->first();

        return Session::create([
            'poste_id' => $poste->id,
            'type_session' => 'ethernet',
            'date_heure_debut' => now(),
            'date_heure_fin_prevue' => now()->addMinutes($dureeMinutes),
            'duree_prevue' => $dureeMinutes,
            'etat' => 'en_cours',
            'montant_initial' => 0,
            'montant_total' => 0,
            'tarif_id' => $tarif?->id,
            'description' => "Session {$type} de {$dureeMinutes} minutes.",
        ]);
    }

    /**
     * Envoie une commande à l'agent C#.
     */
    private function envoyerCommande(Poste $poste, string $commande): array
    {
        if (empty($poste->agent_url)) {
            throw new \Exception("Aucun agent configuré pour ce poste.");
        }

        try {
            $response = Http::withHeaders([
                'X-Agent-Token' => config('mikrotik.agent_token', 'CyberManagerAgent2026_Secret'),
                'Content-Type' => 'application/json'
            ])->timeout(5)->post($poste->agent_url . '/api/commande', [
                'commande' => $commande
            ]);

            if ($response->successful()) {
                return ['succes' => true, 'message' => $response->json('message')];
            }

            return ['succes' => false, 'message' => 'Réponse invalide de l\'agent.'];
        } catch (\Throwable $e) {
            return ['succes' => false, 'message' => $e->getMessage()];
        }
    }

    public function verrouillerPoste(Poste $poste): array
    {
        return $this->envoyerCommande($poste, 'LOCK');
    }

    public function deverrouillerPoste(Poste $poste): array
    {
        return $this->envoyerCommande($poste, 'UNLOCK');
    }

    public function pausePoste(Poste $poste): array
    {
        // Pour l'instant, on utilise LOCK comme pause. On pourra ajouter PAUSE plus tard.
        return $this->envoyerCommande($poste, 'LOCK');
    }

    public function arreterPoste(Poste $poste): array
    {
        return $this->envoyerCommande($poste, 'LOCK');
    }

 *
 * POLITIQUE « coupure » (voir ResilienceService et ParametresResilience) :
 *  - le temps où le poste (ou le serveur) était injoignable n'est JAMAIS facturé :
 *    la session est figée rétroactivement au dernier signe de vie (suspendre() avec un instant passé) ;
 *  - au retour du poste : reprise par le client (défaut), automatique, ou décision du gérant ;
 *  - si personne ne reprend dans les délais réglés : clôture facturée jusqu'à l'instant du gel.
 *
 * MODE DÉGRADÉ : aucune méthode réseau ne lève d'exception ; la session s'ouvre / se ferme quand même.
 */
class PosteSessionService
{
    public function __construct(
        private MikroTikService $mikroTik,
        private TarifService $tarifs
    ) {}

    // ------------------------------------------------------------------
    // Cycle de vie de la session (base de données)
    // ------------------------------------------------------------------

    public function creerSession(Poste $poste, string $mode, ?int $duree): Session
    {
        try {
            $tarifId = $this->tarifs->tarifStandard()->id;
        } catch (\Throwable) {
            $tarifId = null;
        }

        $session = Session::create([
            'poste_id'              => $poste->id,
            'type_session'          => 'ethernet',
            'date_heure_debut'      => now(),
            'date_heure_fin_prevue' => $duree ? now()->addMinutes($duree) : null,
            'duree_prevue'          => $duree ?? 0,
            'etat'                  => 'en_cours',
            'tarif_id'              => $tarifId,
            'montant_initial'       => 0,
            'montant_total'         => 0,
            'description'           => "Session {$mode}" . ($duree ? " de {$duree} minutes." : '.'),
        ]);

        $session->mode_facturation = $mode;
        $session->save();

        return $session;
    }

    /**
     * Montant (Ar) de la session. Le mode « libre » n'est jamais facturé.
     *
     * @param int|null $secondes Durée facturable à utiliser ; null = durée consommée à l'instant T
     */
    public function montant(Session $session, ?int $secondes = null): int
    {
        if (($session->mode_facturation ?? 'forfait') === 'libre') {
            return 0;
        }

        try {
            $tarif   = $session->tarif ?: $this->tarifs->tarifStandard();
            $minutes = intdiv($secondes ?? $session->secondesConsommees(), 60);

            return $this->tarifs->montantTempsReel($minutes, $tarif);
        } catch (\Throwable) {
            return 0;
        }
    }

    /**
     * Suspend la session active : chrono figé, poste « suspendu ».
     * Base de données uniquement (à appeler dans une transaction) ; la coupure réseau
     * se fait ensuite via deconnecterReseau().
     *
     * @param CarbonInterface|null $a   Instant de gel (passé autorisé : gel rétroactif). Défaut : maintenant.
     *                                  Borné : jamais avant le début/la dernière reprise, jamais dans le futur.
     * @param string               $motif Session::PAUSE_* (manuel, coupure_poste, coupure_serveur, horloge)
     *
     * @return bool false si le poste n'a aucune session active
     */
    public function suspendre(
        Poste $poste,
        ?CarbonInterface $a = null,
        string $motif = Session::PAUSE_MANUEL,
        ?int $incidentId = null
    ): bool {
        $session = $poste->sessionActive();
        if ($session === null) {
            return false;
        }

        if ($session->etat === 'en_cours') {
            $session->update([
                'etat'                  => 'en_pause',
                'date_heure_suspension' => $this->bornerGel($session, $a),
                'motif_pause'           => $motif,
                'incident_id'           => $incidentId ?? $session->incident_id,
                'poste_revenu_at'       => null,
            ]);
        }
        $poste->update(['etat' => Poste::ETAT_SUSPENDU]);

        return true;
    }

    /**
     * Reprend la session active (si elle était en pause) et passe le poste « occupé ».
     * Le temps de pause n'est pas facturé ; pour un forfait, la fin prévue est repoussée d'autant.
     * Base de données uniquement ; la reconnexion réseau se fait ensuite.
     *
     * @return bool false si le poste n'a aucune session active
     */
    public function reprendre(Poste $poste): bool
    {
        $session = $poste->sessionActive();
        if ($session === null) {
            return false;
        }

        if ($session->etat === 'en_pause') {
            $maintenant = now();
            $pause = $session->date_heure_suspension
                ? max(0, $maintenant->getTimestamp() - $session->date_heure_suspension->getTimestamp())
                : 0;
            $coupure    = $session->estInterrompueParCoupure();
            $incidentId = $session->incident_id;

            $maj = [
                'etat'                  => 'en_cours',
                'date_heure_reprise'    => $maintenant,
                'duree_suspension'      => (int) $session->duree_suspension + $pause,
                'date_heure_suspension' => null,
                'motif_pause'           => null,
                'poste_revenu_at'       => null,
            ];
            if ($coupure) {
                // Sous-ensemble de duree_suspension, uniquement pour l'affichage « temps non facturé »
                $maj['duree_coupure_secondes'] = (int) $session->duree_coupure_secondes + $pause;
            }
            if ($session->date_heure_fin_prevue !== null && $pause > 0) {
                $maj['date_heure_fin_prevue'] = $session->date_heure_fin_prevue->copy()->addSeconds($pause);
            }

            $session->update($maj);
            $this->actualiserIncidentPoste($incidentId);
        }

        $poste->update(['etat' => Poste::ETAT_OCCUPE]);

        return true;
    }

    /**
     * Clôture la session active : calcule et fige le montant, poste « désactivé ».
     * Base de données uniquement (à appeler dans une transaction) ; la coupure réseau
     * se fait ensuite via deconnecterReseau().
     *
     * @param CarbonInterface|null $arretA Instant (passé) où la consommation s'est réellement arrêtée
     *                                     (expiration d'un forfait). Ignoré si la session est en pause
     *                                     (son chrono est déjà figé).
     */
    public function cloturer(Poste $poste, string $motif = 'client', ?CarbonInterface $arretA = null): ?Session
    {
        $session = $poste->sessionActive();
        if ($session === null) {
            return null;
        }

        $brut     = $session->secondesConsommees();   // avant le changement d'état
        $secondes = $brut;
        $fin      = now();

        // Session en pause : la pause en cours doit être ajoutée à duree_suspension, sinon
        // secondesConsommees() d'une session terminée la compterait comme consommée.
        $pauseEnCours = 0;
        $coupure      = false;
        if ($session->etat === 'en_pause' && $session->date_heure_suspension !== null) {
            $pauseEnCours = max(0, $fin->getTimestamp() - $session->date_heure_suspension->getTimestamp());
            $coupure      = $session->estInterrompueParCoupure();
        }

        if ($arretA !== null && $session->etat === 'en_cours' && $session->date_heure_debut !== null) {
            $arret = $arretA->copy();
            if ($arret->lt($session->date_heure_debut)) {
                $arret = $session->date_heure_debut->copy();
            }
            if ($arret->lt($fin)) {
                // Le chrono comptait en continu : on retire le temps écoulé depuis l'arrêt réel
                $secondes = max(0, $brut - ($fin->getTimestamp() - $arret->getTimestamp()));
                $fin      = $arret;
            }
        }

        $montant = $this->montant($session, $secondes);

        $session->update([
            'etat'                   => 'terminee',
            'date_heure_fin_reelle'  => $fin,
            'motif_fin'              => $motif,
            'duree_consommee'        => intdiv($secondes, 60),
            'montant_consomme'       => $montant,
            'montant_total'          => $montant,
            'duree_suspension'       => (int) $session->duree_suspension + $pauseEnCours,
            'duree_coupure_secondes' => (int) $session->duree_coupure_secondes + ($coupure ? $pauseEnCours : 0),
        ]);

        $this->actualiserIncidentPoste($session->incident_id);

        // Fin de session (fermeture, expiration, abandon) : le poste passe en « Désactivé ».
        // Il ne redevient disponible que lorsque le gérant clique sur « Activer ».
        $poste->update(['etat' => Poste::ETAT_DESACTIVE]);

        return $session;
    }

    // ------------------------------------------------------------------
    // Expiration des forfaits
    // ------------------------------------------------------------------

    /** Vrai si la session est un forfait en cours dont le temps est épuisé. */
    public function sessionExpiree(Session $session): bool
    {
        return $session->etat === 'en_cours'
            && ($session->mode_facturation ?? 'forfait') === 'forfait'
            && (int) $session->duree_prevue > 0
            && $session->secondesConsommees() >= (int) $session->duree_prevue * 60;
    }

    /**
     * Clôture la session du poste si son forfait est épuisé, puis coupe Internet.
     * Idempotent et sûr en concurrence : appelable depuis le heartbeat ET depuis la maintenance.
     */
    public function expirerSiNecessaire(Poste $poste): ?Session
    {
        $apercu = $poste->sessionActive();
        if ($apercu === null || !$this->sessionExpiree($apercu)) {
            return null;
        }

        if (ResilienceService::horlogeBloquee()) {
            return null;   // horloge suspecte : aucun calcul automatique
        }

        $session = DB::transaction(function () use ($poste) {
            $p = Poste::whereKey($poste->id)->lockForUpdate()->first();
            $s = $p?->sessionActive();
            if ($s === null || !$this->sessionExpiree($s)) {
                return null;
            }

            // Instant exact où le forfait a été épuisé (en tenant compte des temps non facturés)
            $deductions = (int) $s->duree_suspension + (int) $s->duree_hors_ligne_secondes + (int) $s->duree_pause_secondes;
            $arret      = $s->date_heure_debut->copy()->addSeconds((int) $s->duree_prevue * 60 + $deductions);

            return $this->cloturer($p, 'expiration', $arret->isFuture() ? null : $arret);
        });

        if ($session === null) {
            return null;
        }

        Log::info("Forfait expiré : session #{$session->id} du poste {$poste->nom_poste} clôturée ({$session->montant_total} Ar).");
        $this->deconnecterReseau($poste);

        return $session;
    }

    // ------------------------------------------------------------------
    // Poste injoignable / retour en ligne
    // ------------------------------------------------------------------

    /**
     * Micro-coupure : le poste redonne signe de vie après une panne plus longue que le délai
     * d'injoignabilité mais sans avoir été figé. Le temps d'injoignabilité n'est pas facturé.
     *
     * @return int Secondes déduites
     */
    public function deduireTempsHorsLigne(Poste $poste, ?CarbonInterface $dernierSigne): int
    {
        if ($dernierSigne === null) {
            return 0;
        }

        return DB::transaction(function () use ($poste, $dernierSigne) {
            $p = Poste::whereKey($poste->id)->lockForUpdate()->first();
            $s = $p?->sessionActive();
            if ($s === null || $s->etat !== 'en_cours') {
                return 0;   // une session figée n'est jamais déduite deux fois
            }

            // La panne ne peut pas avoir commencé avant le début ou la dernière reprise de la session
            $debutPanne = max(
                $dernierSigne->getTimestamp(),
                $s->date_heure_debut?->getTimestamp() ?? 0,
                $s->date_heure_reprise?->getTimestamp() ?? 0
            );
            $panne = now()->getTimestamp() - $debutPanne;
            if ($panne <= Poste::delaiHorsLigne()) {
                return 0;
            }

            $s->update(['duree_hors_ligne_secondes' => (int) $s->duree_hors_ligne_secondes + $panne]);
            Log::info("Poste {$poste->nom_poste} de retour après {$panne} s : temps de panne déduit de la session #{$s->id}.");

            return $panne;
        });
    }

    /**
     * Le poste donne signe de vie alors que sa session est figée par une coupure :
     * enregistre son retour (départ du délai d'abandon) et applique la politique de reprise.
     * Idempotent : n'agit qu'au premier signe de vie après le gel.
     *
     * @return bool vrai si la session active du poste est (ou était) figée par une coupure
     */
    public function traiterRetourSiGele(Poste $poste): bool
    {
        $session = $poste->sessionActive();
        if ($session === null || !$session->estInterrompueParCoupure()) {
            return false;
        }
        if ($session->poste_revenu_at !== null) {
            return true;   // déjà traité
        }

        $session->update(['poste_revenu_at' => now()]);

        // Fin de l'incident d'un poste seul : la durée de la coupure est connue
        if ($session->incident_id) {
            $incident = Incident::find($session->incident_id);
            if ($incident && $incident->type === Incident::TYPE_COUPURE_POSTE && $incident->fin_at === null) {
                $incident->update([
                    'fin_at'         => now(),
                    'duree_secondes' => max(0, now()->getTimestamp() - $incident->debut_at->getTimestamp()),
                ]);
            }
        }

        if (ParametresResilience::politiqueReprise() === ParametresResilience::POLITIQUE_AUTO
            && !ResilienceService::horlogeBloquee()) {
            $repris = DB::transaction(function () use ($poste) {
                $p = Poste::whereKey($poste->id)->lockForUpdate()->first();
                return $p !== null && $this->reprendre($p);
            });

            if ($repris) {
                Log::info("Poste {$poste->nom_poste} : session reprise automatiquement après la coupure (politique « auto »).");
                $this->connecterReseau($poste);
            }
        }

        return true;
    }

    /**
     * Retour en ligne d'un poste (redémarrage, coupure) : gère une session figée, sinon déduit
     * la panne, applique une éventuelle expiration, puis rétablit Internet si la session tourne.
     */
    public function traiterRetourEnLigne(Poste $poste, ?CarbonInterface $dernierSigne): void
    {
        if ($this->traiterRetourSiGele($poste)) {
            return;
        }

        $this->deduireTempsHorsLigne($poste, $dernierSigne);

        if ($this->expirerSiNecessaire($poste) !== null) {
            return;
        }

        $session = $poste->sessionActive();
        if ($session !== null && $session->etat === 'en_cours') {
            $this->connecterReseau($poste);
        }
    }

    /**
     * Clôture une session figée par une coupure si personne ne l'a reprise dans les délais :
     *  - poste revenu : abandon_apres_retour_minutes après son retour ;
     *  - poste jamais revenu : abandon_poste_absent_minutes après le gel.
     * Facturée jusqu'à l'instant du gel ; Internet reste coupé. Politique « gérant » : jamais d'abandon automatique.
     */
    public function abandonnerSiNecessaire(Poste $poste): ?Session
    {
        if (ParametresResilience::politiqueReprise() === ParametresResilience::POLITIQUE_GERANT
            || ResilienceService::horlogeBloquee()) {
            return null;
        }

        $session = DB::transaction(function () use ($poste) {
            $p = Poste::whereKey($poste->id)->lockForUpdate()->first();
            $s = $p?->sessionActive();
            if ($s === null || !$s->estInterrompueParCoupure()) {
                return null;
            }

            if ($s->poste_revenu_at !== null) {
                $delai = ParametresResilience::abandonApresRetourMinutes();
                if ($delai <= 0 || $s->poste_revenu_at->gt(now()->subMinutes($delai))) {
                    return null;
                }
            } else {
                $delai = ParametresResilience::abandonPosteAbsentMinutes();
                $gel   = $s->date_heure_suspension;
                if ($delai <= 0 || $gel === null || $gel->gt(now()->subMinutes($delai))) {
                    return null;
                }
            }

            return $this->cloturer($p, 'coupure');
        });

        if ($session === null) {
            return null;
        }

        Log::warning("Poste {$poste->nom_poste} : session #{$session->id} clôturée après la coupure sans reprise ({$session->montant_total} Ar, facturée jusqu'au gel).");
        $this->deconnecterReseau($poste);

        return $session;
    }

    // ------------------------------------------------------------------
    // Réseau : AccesInternetService est l'unique autorité (liaison par adresse MAC)
    // ------------------------------------------------------------------

    /** Le compte HotSpot existe-t-il sur MikroTik ? (conservé pour l'existant) */
    public function compteHotspotExiste(string $username): ?bool
    {
        if (!$this->mikrotikActif()) {
            return null;
        }

        try {
            return $this->mikroTik->findHotspotUser($username) !== null;
        } catch (\Throwable $e) {
            Log::warning('MikroTik : vérification du compte HotSpot impossible (' . $e->getMessage() . ').');
            return null;
        }
    }

    public function mikrotikActif(): bool
    {
        return (bool) config('mikrotik.enabled', true);
    }

    /** @return bool|null true = autorisé, false = échec routeur (mode dégradé), null = aucun routeur */
    public function connecterReseau(Poste $poste): ?bool
    {
        return app(AccesInternetService::class)->autoriser($poste);
    }

    /** @return bool|null true = coupé, false = échec routeur (mode dégradé), null = aucun routeur */
    public function deconnecterReseau(Poste $poste): ?bool
    {
        return app(AccesInternetService::class)->couper($poste);
    }

    /**
     * Aligne le routeur sur l'état réel des sessions (auto-réparation).
     *
     * @return array{connectes:int, coupes:int, erreurs:int}
     */
    public function reconcilierReseau(): array
    {
        return app(AccesInternetService::class)->reconcilier();
    }

    // ------------------------------------------------------------------
    // Outils internes
    // ------------------------------------------------------------------

    /** Instant de gel borné : pas avant le début / la dernière reprise, pas dans le futur. */
    private function bornerGel(Session $session, ?CarbonInterface $a): CarbonInterface
    {
        $maintenant = now();
        $gel        = $a !== null ? $a->copy() : $maintenant->copy();

        if ($gel->gt($maintenant)) {
            $gel = $maintenant->copy();
        }
        if ($session->date_heure_debut !== null && $gel->lt($session->date_heure_debut)) {
            $gel = $session->date_heure_debut->copy();
        }
        if ($session->date_heure_reprise !== null && $gel->lt($session->date_heure_reprise)) {
            $gel = $session->date_heure_reprise->copy();
        }

        return $gel;
    }

    /** L'incident d'un poste seul se résout tout seul quand plus aucune session n'est figée. */
    private function actualiserIncidentPoste(?int $incidentId): void
    {
        if ($incidentId === null) {
            return;
        }

        $incident = Incident::find($incidentId);
        if ($incident === null || $incident->type !== Incident::TYPE_COUPURE_POSTE || !$incident->estOuvert()) {
            return;
        }

        if (!Session::where('incident_id', $incident->id)->where('etat', 'en_pause')->exists()) {
            $incident->acquitter(null);
        }
    }
}
